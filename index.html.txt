  <script>
    // 30 Kişilik Sınıf Listesi
    const students = [
      "Ahmet Yılmaz", "Ayşe Demir", "Mehmet Kaya", "Fatma Çelik", "Mustafa Şahin", 
      "Zeynep Yıldız", "Ali Yıldırım", "Elif Öztürk", "Hasan Aydın", "Hatice Özdemir", 
      "İbrahim Arslan", "Merve Doğan", "Hüseyin Kılıç", "Esra Aslan", "Osman Çetin", 
      "Büşra Kara", "İsmail Koç", "Kübra Kurt", "Yusuf Özkan", "Meryem Şimşek", 
      "Ömer Polat", "Yasemin Öz", "Ramazan Korkmaz", "Tuğba Erdoğan", "Halil Yavuz", 
      "Emine Can", "Süleyman Yücel", "Gökçe Yalçın", "Burak Güneş", "Ceren Bozkurt"
    ];

    let currentStudent = "";
    let studentData = JSON.parse(localStorage.getItem("yks_student_data")) || {};

    // Sayfa Yüklendiğinde
    window.onload = () => {
      document.getElementById('entryDate').valueAsDate = new Date();
      renderRoster();
      renderReport();
    };

    // Ekran Değiştirme
    function switchView(viewId) {
      document.querySelectorAll('.view').forEach(el => el.classList.remove('active'));
      document.querySelectorAll('.nav-btn').forEach(el => el.classList.remove('active'));
      
      document.getElementById(viewId).classList.add('active');
      if (viewId === 'reportView') renderReport();
    }

    // Öğrenci Kartlarını Oluşturma
    function renderRoster() {
      const grid = document.getElementById('studentGrid');
      grid.innerHTML = "";
      students.forEach((name, index) => {
        const card = document.createElement('div');
        card.className = 'student-card';
        card.onclick = () => openStudentPortal(name);
        card.innerHTML = `
          <div class="avatar">${name.charAt(0)}</div>
          <div style="font-weight:600;">${name}</div>
          <div style="font-size:0.8rem; color:var(--text-muted);">No: ${101 + index}</div>
        `;
        grid.appendChild(card);
      });
    }

    // Öğrenci Sayfasına Geçiş
    function openStudentPortal(name) {
      currentStudent = name;
      document.getElementById('selectedStudentName').innerText = name;
      switchView('entryView');
    }

    // Veri Kaydetme
    document.getElementById('questionForm').onsubmit = (e) => {
      e.preventDefault();
      const date = document.getElementById('entryDate').value;
      
      const subjects = ['turkce', 'matematik', 'tarih', 'cografya', 'felsefe', 'din', 'fen'];
      let dayTotal = 0;

      subjects.forEach(sub => {
        const val = parseInt(document.getElementById(`sub_${sub}`).value) || 0;
        dayTotal += val;
      });

      if (!studentData[currentStudent]) studentData[currentStudent] = [];
      studentData[currentStudent].push({ date, total: dayTotal });

      localStorage.setItem("yks_student_data", JSON.stringify(studentData));
      alert("Soru sayıları başarıyla kaydedildi!");
      document.getElementById('questionForm').reset();
      document.getElementById('entryDate').valueAsDate = new Date();
      switchView('rosterView');
    };

    // Öğretmen Rapor Tablosunu Oluşturma
    function renderReport() {
      const tbody = document.getElementById('reportTableBody');
      tbody.innerHTML = "";

      students.forEach((name, index) => {
        const records = studentData[name] || [];
        const totalSolved = records.reduce((acc, curr) => acc + curr.total, 0);

        const tr = document.createElement('tr');
        tr.innerHTML = `
          <td>${101 + index}</td>
          <td><strong>${name}</strong></td>
          <td>${totalSolved} Soru</td>
        `;
        tbody.appendChild(tr);
      });
    }
  </script>
